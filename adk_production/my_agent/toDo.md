package main

import (
	"fmt"
	"math"
)

// SimpleLinearRegression represents our model: y = mx + b
type SimpleLinearRegression struct {
	Slope     float64 // m
	Intercept float64 // b
}

// Fit trains the model using Ordinary Least Squares
func (lr *SimpleLinearRegression) Fit(x, y []float64) {
	n := float64(len(x))
	var sumX, sumY float64
	for i := 0; i < len(x); i++ {
		sumX += x[i]
		sumY += y[i]
	}
	meanX := sumX / n
	meanY := sumY / n

	var numerator, denominator float64
	for i := 0; i < len(x); i++ {
		numerator += (x[i] - meanX) * (y[i] - meanY)
		denominator += (x[i] - meanX) * (x[i] - meanX)
	}

	lr.Slope = numerator / denominator
	lr.Intercept = meanY - (lr.Slope * meanX)
}

// Predict calculates the estimated value for a given input
func (lr *SimpleLinearRegression) Predict(x float64) float64 {
	return lr.Slope*x + lr.Intercept
}

// MeanSquaredError computes MSE: (1/n) * sum((actual - predicted)^2)
func MeanSquaredError(actual, predicted []float64) float64 {
	var sum float64
	for i := range actual {
		diff := actual[i] - predicted[i]
		sum += diff * diff
	}
	return sum / float64(len(actual))
}

// R2Score measures goodness-of-fit
func R2Score(actual, predicted []float64) float64 {
	var sum float64
	for _, val := range actual {
		sum += val
	}
	mean := sum / float64(len(actual))

	var ssRes, ssTot float64
	for i := range actual {
		ssRes += math.Pow(actual[i]-predicted[i], 2)
		ssTot += math.Pow(actual[i]-mean, 2)
	}
	return 1 - (ssRes / ssTot)
}

func main() {
	// ---------------------------------------------------------
	// Step 1: Create dataset (split into Train and Test sets)
	// ---------------------------------------------------------
	// Training set (Hours studied -> Exam score)
	xTrain := []float64{1.0, 2.0, 2.5, 3.0, 3.5, 5.0, 7.0, 8.0, 9.0, 10.0}
	yTrain := []float64{35.0, 42.0, 48.0, 50.0, 55.0, 68.0, 80.0, 88.0, 91.0, 98.0}

	// Test set (held-out data)
	xTest := []float64{4.0, 6.0}
	yTest := []float64{62.0, 74.0}

	// ---------------------------------------------------------
	// Step 2: Initialize and train the model
	// ---------------------------------------------------------
	model := SimpleLinearRegression{}
	model.Fit(xTrain, yTrain)

	// ---------------------------------------------------------
	// Step 3: Evaluate on test data
	// ---------------------------------------------------------
	predictions := make([]float64, len(xTest))
	for i, x := range xTest {
		predictions[i] = model.Predict(x)
	}

	mse := MeanSquaredError(yTest, predictions)
	r2 := R2Score(yTest, predictions)

	fmt.Printf("Model Slope (m): %.2f points per study hour\n", model.Slope)
	fmt.Printf("Model Intercept (b): %.2f baseline score\n", model.Intercept)
	fmt.Printf("Mean Squared Error: %.2f\n", mse)
	fmt.Printf("R² Score: %.2f\n", r2)

	// ---------------------------------------------------------
	// Step 4: Make a new prediction
	// ---------------------------------------------------------
	hours := 6.5
	predictedScore := model.Predict(hours)
	fmt.Printf("\nPredicted score for studying %.1f hours: %.1f / 100\n", hours, predictedScore)
}
